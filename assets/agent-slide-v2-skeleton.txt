  function stack(box, c, id) {
    box.innerHTML = '<div class="cwrap"></div>';
    var wrap = box.firstChild, W = wrap.clientWidth, H = wrap.clientHeight; if (!W || !H) return;
    var svg = el("svg", { width: W, height: H }, wrap), segs = c.segments || [];
    var L = W < 950 ? 150 : 230, R = 28, T = 58, B = 32, iw = W - L - R, ih = H - T - B, band = ih / c.labels.length, thick = Math.min(band * 0.58, 52);
    var lstep = Math.min(220, (W - 20) / Math.max(1, segs.length));
    segs.forEach(function (sg, si) {
      check(id + " " + sg.name, c.labels, sg.values);
      var col = sg.hi ? RED : PAL[si % PAL.length], lx = 10 + si * lstep;
      el("rect", { x: lx, y: 12, width: 20, height: 20, fill: col, rx: 4 }, svg);
      label(svg, lx + 28, 29, sg.name, 20, INK, "start");
    });
    c.labels.forEach(function (lb, i) {
      var cy = T + band * i + band / 2, sum = segs.reduce(function (a, sg) { return a + Number(sg.values[i] || 0); }, 0) || 1, acc = 0;
      label(svg, L - 14, cy + 7, lb, 20, INK, "end");
      segs.forEach(function (sg, si) {
        var val = Number(sg.values[i] || 0), pct = val / sum * 100, w = iw * (val / sum), col = sg.hi ? RED : PAL[si % PAL.length];
        var r = el("rect", { x: L + acc, y: cy - thick / 2, width: Math.max(1, w - 2), height: thick, fill: sg.hi ? mkGrad(svg, RED, ORANGE, 1, 0.94, false) : col, rx: 5 }, svg);
        r.style.cursor = "pointer";
        r.onmousemove = function (e) { tipOn(e, "<b>" + lb + " · " + sg.name + "</b><br>" + fmt(val, c.digits) + (c.unit || "") + " (" + pct.toFixed(1) + "%)"); };
        r.onmouseleave = tipOff;
        if (pct >= 11) label(svg, L + acc + w / 2, cy + 7, pct.toFixed(0) + "%", 20, (!sg.hi && (si % PAL.length) >= 3) ? INK : "#fff", "middle").setAttribute("font-weight", 700);
        acc += w;
      });
    });
  }

  function line(box, c, id) {
    box.innerHTML = '<div class="cwrap"></div>';
    var wrap = box.firstChild, W = wrap.clientWidth, H = wrap.clientHeight; if (!W || !H) return;
    var svg = el("svg", { width: W, height: H }, wrap), all = [];
    c.lines.forEach(function (ln) { check(id + " " + (ln.name || ""), c.labels, ln.values); all = all.concat(ln.values.map(Number)); });
    if (c.target != null) all.push(c.target);
    var mn = Math.min.apply(null, all), mx = Math.max.apply(null, all), s = scale(mn - (mx - mn || 1) * 0.14, mx + (mx - mn || 1) * 0.14);
    var lo = c.min != null ? c.min : s.lo, hi = c.max != null ? c.max : s.hi;
    var L = 100, R = 56, T = c.lines.length > 1 ? 60 : 38, B = 64, iw = W - L - R, ih = H - T - B, n = c.labels.length;
    function X(i) { return L + (n < 2 ? iw / 2 : iw * i / (n - 1)); }
    function Y(v) { return T + ih * (1 - (v - lo) / (hi - lo || 1)); }
    for (var g = Math.ceil(lo / s.step) * s.step; g <= hi + 1e-9; g += s.step) { el("line", { x1: L, x2: L + iw, y1: Y(g), y2: Y(g), stroke: "#EEF0F2" }, svg); label(svg, L - 12, Y(g) + 7, fmt(g, s.step < 1 ? 1 : 0), 20, SUB, "end"); }
    c.labels.forEach(function (lb, i) { label(svg, X(i), H - B + 32, lb, 20, INK, "middle"); });
    if (c.target != null) { el("line", { x1: L, x2: L + iw, y1: Y(c.target), y2: Y(c.target), stroke: RED, "stroke-dasharray": "6 5", "stroke-width": 2.5 }, svg); label(svg, L + 8, Y(c.target) + 26, "목표 " + fmt(c.target, c.digits) + (c.unit || ""), 20, RED, "start").setAttribute("font-weight", 700); }
    c.lines.forEach(function (ln, li) {
      var col = ln.hi ? RED : [ORANGE, "#6C727A", GRAY][Math.min(li, 2)];
      var dLine = ln.values.map(function (v, i) { return (i ? "L" : "M") + X(i) + " " + Y(v); }).join(" ");
      if (li === 0 && n > 1) {
        el("path", { d: dLine + " L" + X(n - 1) + " " + (T + ih) + " L" + X(0) + " " + (T + ih) + " Z", fill: mkGrad(svg, col, ORANGE, 0.22, 0.0, true), "class": "fade" }, svg);
      }
      el("path", { d: dLine, fill: "none", stroke: col, "stroke-width": li === 0 ? 4.5 : 3, pathLength: 1, "class": "draw" }, svg);
      ln.values.forEach(function (v, i) {
        var last = i === n - 1, dot = el("circle", { cx: X(i), cy: Y(v), r: last ? 8 : 6, fill: "#fff", stroke: col, "stroke-width": 3 }, svg);
        dot.style.cursor = "pointer";
        dot.onmousemove = function (e) { tipOn(e, "<b>" + c.labels[i] + "</b><br>" + (ln.name || "") + " " + fmt(v, c.digits) + (c.unit || "") + ((c.notes || [])[i] ? "<br>" + c.notes[i] : "")); };
        dot.onmouseleave = tipOff;
        if (last && li === 0) label(svg, X(i), Y(v) - 18, fmt(v, c.digits) + (c.unit || ""), 20, col, "middle", "fade").setAttribute("font-weight", 700);
      });
      if (c.lines.length > 1) { el("rect", { x: L + li * 240, y: 12, width: 28, height: 6, fill: col }, svg); label(svg, L + li * 240 + 38, 22, ln.name || "", 20, INK, "start"); }
    });
  }

  function sim(box, cfg) {
    if (box.dataset.ready) return;
    box.dataset.ready = "1";
    var vals = {}, inputs = {}, valEls = {};
    cfg.sliders.forEach(function (sl) { vals[sl.key] = sl.value; });
    box.innerHTML = '<div class="sim-in"><div class="sim-presets"></div><div class="sim-rows"></div></div><div class="sim-out"><div class="sim-cards"></div><div class="sim-disc"></div></div>';
    var preBox = box.querySelector(".sim-presets"), rowBox = box.querySelector(".sim-rows"), outBox = box.querySelector(".sim-cards");
    box.querySelector(".sim-disc").textContent = cfg.disclaimer || "※ 가정 기반 시뮬레이션 결과입니다.";
    (cfg.presets || []).forEach(function (pr) {
      var b = document.createElement("button"); b.textContent = pr.name;
      b.onclick = function () {
        for (var k in pr.values) { vals[k] = pr.values[k]; if (inputs[k]) inputs[k].value = pr.values[k]; }
        [].slice.call(preBox.children).forEach(function (x) { x.classList.toggle("on", x === b); });
        update();
      };
      preBox.appendChild(b);
    });
    var rst = document.createElement("button"); rst.textContent = "초기화";
    rst.onclick = function () { cfg.sliders.forEach(function (sl) { vals[sl.key] = sl.value; inputs[sl.key].value = sl.value; }); [].slice.call(preBox.children).forEach(function (x) { x.classList.remove("on"); }); update(); };
    preBox.appendChild(rst);
    cfg.sliders.forEach(function (sl) {
      var r = document.createElement("div"); r.className = "sim-row";
      r.innerHTML = '<div class="sim-top"><span>' + sl.label + '</span><b></b></div><input type="range"><div class="sim-mm"><span>' + fmt(sl.min, sl.digits) + (sl.unit || "") + '</span><span>' + fmt(sl.max, sl.digits) + (sl.unit || "") + '</span></div>';
      var inp = r.querySelector("input"); inp.min = sl.min; inp.max = sl.max; inp.step = sl.step; inp.value = sl.value;
      valEls[sl.key] = r.querySelector("b"); inputs[sl.key] = inp;
      inp.oninput = function () { vals[sl.key] = Number(inp.value); [].slice.call(preBox.children).forEach(function (x) { x.classList.remove("on"); }); update(); };
      rowBox.appendChild(r);
    });
    function update() {
      cfg.sliders.forEach(function (sl) { valEls[sl.key].textContent = fmt(vals[sl.key], sl.digits) + (sl.unit || ""); });
      outBox.innerHTML = "";
      cfg.outputs.forEach(function (o) {
        var dv = 0; for (var k in o.effects) dv += Number(vals[k] || 0) * Number(o.effects[k]);
        var cv = o.base + dv, good = o.target == null ? true : (o.dir === "down" ? cv <= o.target : cv >= o.target);
        var pct = Math.max(0, Math.min(100, (cv - o.min) / (o.max - o.min || 1) * 100));
        var tpct = o.target == null ? -1 : Math.max(0, Math.min(100, (o.target - o.min) / (o.max - o.min || 1) * 100));
        var card = document.createElement("div"); card.className = "sim-card" + (good ? "" : " ng");
        card.innerHTML = '<div class="sim-cl"><div class="sim-ch"><span>' + o.label + '</span>' + (o.target != null ? '<b>' + (good ? "목표 달성 ✓" : "목표 미달") + '</b>' : '') + '</div><div class="sim-val">' + fmt(cv, o.digits) + '<small>' + (o.unit || "") + '</small></div></div><div class="sim-cr"><div class="sim-diff">현재 ' + fmt(o.base, o.digits) + (o.unit || "") + ' 대비 ' + (dv > 0 ? "+" : "") + fmt(dv, o.digits) + (o.unit || "") + (o.target != null ? ' · 목표 ' + fmt(o.target, o.digits) + (o.unit || "") : '') + '</div><div class="sim-bar"><div class="sim-fill" style="width:' + pct + '%"></div>' + (tpct >= 0 ? '<div class="sim-tgt" style="left:' + tpct + '%"></div>' : '') + '</div><div class="sim-note">' + (o.note || "") + '</div></div>';
        outBox.appendChild(card);
      });
    }
    update();
  }

  function spark(box) {
    var vals = (box.getAttribute("data-trend") || "").split(",").map(Number), W = box.clientWidth, H = box.clientHeight;
    if (vals.length < 2 || !W || !H) return;
    box.innerHTML = "";
    var mn = Math.min.apply(null, vals), mx = Math.max.apply(null, vals), svg = el("svg", { width: W, height: H }, box);
    var sc = box.closest(".ng") ? RED : ORANGE;
    var sd = vals.map(function (v, i) { return (i ? "L" : "M") + (4 + (W - 8) * i / (vals.length - 1)) + " " + (6 + (H - 12) * (1 - (v - mn) / (mx - mn || 1))); }).join(" ");
    el("path", { d: sd + " L" + (W - 4) + " " + (H - 2) + " L4 " + (H - 2) + " Z", fill: mkGrad(svg, sc, sc, 0.22, 0.0, true), "class": "fade" }, svg);
    el("path", { d: sd, fill: "none", stroke: sc, "stroke-width": 3.5, pathLength: 1, "class": "draw" }, svg);
  }
  function countUp(v) {
    var to = Number(v.getAttribute("data-v")), d = Number(v.getAttribute("data-d") || 0), u = v.getAttribute("data-u") || "", st = null;
    function step(t) { if (!st) st = t; var k = Math.min(1, (t - st) / 750); v.innerHTML = fmt(to * (1 - Math.pow(1 - k, 3)), d) + "<small>" + u + "</small>"; if (k < 1) requestAnimationFrame(step); }
    requestAnimationFrame(step);
  }

  window.drawSlide = function (slide) {
    problems = [];
    [].slice.call(slide.querySelectorAll(".chart")).forEach(function (box) {
      var c = C[box.id];
      if (!c) { problems.push(box.id + ": 그래프 데이터가 없습니다"); box.textContent = "그래프 데이터 없음: " + box.id; return; }
      try { ({ line: line, delta: delta, combo: combo, stack: stack }[c.type] || bar)(box, c, box.id); }
      catch (e) { problems.push(box.id + ": 그래프를 그리지 못했습니다"); box.textContent = "그래프 오류: " + box.id; }
    });
    [].slice.call(slide.querySelectorAll(".sim")).forEach(function (box) {
      var cfg = S[box.id]; if (!cfg) { problems.push(box.id + ": 시뮬레이션 데이터가 없습니다"); return; }
      try { sim(box, cfg); } catch (e) { problems.push(box.id + ": 시뮬레이션 오류"); }
    });
    [].slice.call(slide.querySelectorAll(".spark")).forEach(spark);
    [].slice.call(slide.querySelectorAll(".kpi .v")).forEach(countUp);
    report();
  };

  if (typeof go === "function") go(cur);
  else {
    all.forEach(window.drawSlide);

    box.style.display = "block";
  }
